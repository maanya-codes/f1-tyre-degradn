│   ├── cota_2023-2025_raw.csv      # Target: Thermal & strategic variance

    ├── 2_aggression_clustering.ipynb # Groups stints into "Aggressive" vs "Conservative"
    └── 3_breakeven_threshold.ipynb # Models cumulative time gained vs pit-stop penalty time
